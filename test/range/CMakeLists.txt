cmake_minimum_required(VERSION 3.10)
project(z_scope C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra")

option(USE_LINUX_IMPL "Use Linux implementation" OFF)
option(USE_WINDOWS_IMPL "Use Windows implementation" OFF)
option(BUILD_PLATFORM_DEMO "Build the standalone platform demo executable" OFF)

if(USE_LINUX_IMPL)
    add_definitions(-DPLATFORM_LINUX)
    message(STATUS "Using Linux implementation")
elseif(USE_WINDOWS_IMPL)
    add_definitions(-DPLATFORM_WINDOWS)
    message(STATUS "Using Windows implementation")
else()
    if(CMAKE_SYSTEM_NAME MATCHES "Linux")
        add_definitions(-DPLATFORM_LINUX)
        message(STATUS "Auto-detected Linux platform")
    elseif(CMAKE_SYSTEM_NAME MATCHES "Windows")
        add_definitions(-DPLATFORM_WINDOWS)
        message(STATUS "Auto-detected Windows platform")
    else()
        add_definitions(-DPLATFORM_LINUX)
        message(STATUS "Defaulting to Linux implementation")
    endif()
endif()

include_directories(${PROJECT_SOURCE_DIR}/include)
include_directories(${PROJECT_SOURCE_DIR}/internal)

file(GLOB COMMON_SOURCES "src/*.c")
list(FILTER COMMON_SOURCES EXCLUDE REGEX ".*platform_.*\\.c$")
list(FILTER COMMON_SOURCES EXCLUDE REGEX ".*demo_platform\\.c$")
file(GLOB_RECURSE INTERNAL_SOURCES "internal/*.c")

if(CMAKE_SYSTEM_NAME MATCHES "Linux" OR USE_LINUX_IMPL)
    set(PLATFORM_IMPL "src/platform_linux.c")
    message(STATUS "Using Linux platform implementation")
elseif(CMAKE_SYSTEM_NAME MATCHES "Windows" OR USE_WINDOWS_IMPL)
    set(PLATFORM_IMPL "src/platform_windows.c")
    message(STATUS "Using Windows platform implementation")
endif()

add_executable(z_scope ${COMMON_SOURCES} ${INTERNAL_SOURCES} ${PLATFORM_IMPL})
if(BUILD_PLATFORM_DEMO)
    add_executable(demo_platform src/demo_platform.c ${PLATFORM_IMPL})
endif()

if(CMAKE_SYSTEM_NAME MATCHES "Linux" OR USE_LINUX_IMPL)
    target_link_libraries(z_scope pthread)
    if(BUILD_PLATFORM_DEMO)
        target_link_libraries(demo_platform pthread)
    endif()
elseif(CMAKE_SYSTEM_NAME MATCHES "Windows" OR USE_WINDOWS_IMPL)
endif()
