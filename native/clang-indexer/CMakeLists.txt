cmake_minimum_required(VERSION 3.16)
project(zscope_clang_indexer LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
if(LLVM_PACKAGE_VERSION VERSION_LESS 18)
  message(FATAL_ERROR "zscope-clang-indexer requires an LLVM/Clang SDK version 18 or newer")
endif()
add_executable(zscope-clang-indexer main.cpp)
set(ARGUMENT_POLICY "${CMAKE_CURRENT_SOURCE_DIR}/../../zscope/retrieval/compiler_argument_policy.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ARGUMENT_POLICY}")
file(READ "${ARGUMENT_POLICY}" ARGUMENT_POLICY_JSON)
file(SHA256 "${ARGUMENT_POLICY}" ARGUMENT_POLICY_SHA)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CompilerPolicy.h"
  "#pragma once\nstatic constexpr char ArgumentPolicyJSON[] = R\"policy(${ARGUMENT_POLICY_JSON})policy\";\nstatic constexpr char ArgumentPolicySHA[] = \"${ARGUMENT_POLICY_SHA}\";\n")
target_include_directories(zscope-clang-indexer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(zscope-clang-indexer PRIVATE cxx_std_17)
target_include_directories(zscope-clang-indexer SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
target_compile_definitions(zscope-clang-indexer PRIVATE ${LLVM_DEFINITIONS})
if(TARGET clang-cpp)
  target_link_libraries(zscope-clang-indexer PRIVATE clang-cpp LLVM)
else()
  target_link_libraries(zscope-clang-indexer PRIVATE clangTooling clangIndex clangFrontend clangAnalysis clangAST clangLex clangBasic LLVM)
endif()
install(TARGETS zscope-clang-indexer RUNTIME DESTINATION bin)
